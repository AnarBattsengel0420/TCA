% DIAGRAM 2: Hybrid Search Score Fusion Pipeline
% Use in §4.3 - Search and Retrieval Section
% ===================================================================
\begin{tikzpicture}[node distance=2.2cm, auto]
  % Styles
  \tikzstyle{search_box} = [rectangle, draw, rounded corners, fill=cyan!20, 
                            minimum width=2cm, minimum height=0.8cm, thick]
  \tikzstyle{process_box} = [rectangle, draw, rounded corners, fill=blue!15, 
                             minimum width=2.1cm, minimum height=0.7cm, thick]
  \tikzstyle{fusion_box} = [rectangle, draw, rounded corners, fill=purple!15, 
                            minimum width=2.3cm, minimum height=0.8cm, thick]
  \tikzstyle{filter_box} = [diamond, draw, fill=yellow!25, thick]
  \tikzstyle{output_box} = [rectangle, draw, rounded corners, fill=green!20, 
                            minimum width=2.5cm, minimum height=0.8cm, thick]
  
  % Search query input
  \node [rectangle, draw, rounded corners, fill=blue!20, 
         minimum width=2.5cm, minimum height=0.8cm, thick] (query) 
    {Дараахын асуулга};
  
  % Three parallel search branches
  \node [search_box, below of=query, xshift=-3.5cm] (vector) 
    {FAISS Vector};
  \node [search_box, below of=query, xshift=0cm] (bm25) 
    {BM25 Статик};
  \node [search_box, below of=query, xshift=3.5cm] (keyword) 
    {Түлхүүр үг};
  
  % Score outputs
  \node [process_box, below of=vector] (sv) 
    {Векторын скор};
  \node [process_box, below of=bm25] (sb) 
    {BM25 скор};
  \node [process_box, below of=keyword] (sk) 
    {Түлхүүрийн скор};
  
  % Fusion formula
  \node [fusion_box, below of=bm25, yshift=-2.2cm] (fusion) 
    {Жин нэмэх:\\$w_v \cdot s_v + w_b \cdot s_b + w_k \cdot s_k$};
  
  % Confidence threshold
  \node [filter_box, below of=fusion] (threshold) 
    {Итгэл > 0.65?};
  
  % Output
  \node [output_box, below of=threshold, yshift=-1.2cm] (results) 
    {Эрэмбэлэгдсэн үр дүн};
  
  % Rejected path
  \node [process_box, below of=threshold, xshift=-2.5cm] (reject) 
    {Хаях};
  
  % Connections
  \draw [->] (query) -- (vector);
  \draw [->] (query) -- (bm25);
  \draw [->] (query) -- (keyword);
  
  \draw [->] (vector) -- (sv);
  \draw [->] (bm25) -- (sb);
  \draw [->] (keyword) -- (sk);
  
  \draw [->] (sv) -- (fusion);
  \draw [->] (sb) -- (fusion);
  \draw [->] (sk) -- (fusion);
  
  \draw [->] (fusion) -- (threshold);
  
  \draw [->] (threshold) -- node[right, font=\small] {Тийм} (results);
  \draw [->] (threshold) -- node[left, font=\small] {Үгүй} (reject);
  
\end{tikzpicture}
